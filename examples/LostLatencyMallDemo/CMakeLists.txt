cmake_minimum_required(VERSION 3.10)
project(lost_latency_mall)
set(APP_NAME "lost_latency_mall")
set(DEVELOPER_NAME "VXPstore")
set(RAM "900")
set(IMSI "91234567890" CACHE STRING "IMSI")
set(API "File ProMng")
set(APPID "3000790" CACHE STRING "App id")
set(CERTID "1" CACHE STRING "Certificate id")
set(CERT "none" CACHE STRING "Certificate")
option(BACKGROUND "Background mode" OFF)
set(MRE_SDK "$ENV{MRE_SDK}" CACHE STRING "MRE SDK")
set(VXPE_PYTHON "$ENV{VXPE_PYTHON}" CACHE FILEPATH "Python")
set(VXPE_SDK_TOOLS "$ENV{VXPE_SDK_TOOLS}" CACHE PATH "VXPEngine SDK tools")
if(NOT VXPE_PYTHON)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(VXPE_PYTHON "${Python3_EXECUTABLE}")
endif()
if(NOT EXISTS "${VXPE_SDK_TOOLS}/vxp_pack.py")
  message(FATAL_ERROR "Missing VXPEngine SDK tools; set -DVXPE_SDK_TOOLS=<engine/coremre/tools>")
endif()
if(BACKGROUND)
  add_definitions(-DSUPPORT_BG)
endif()
include(.vxpe/cmake/MreUtils.cmake)
if(DEFINED COREMRE_PACKAGE_DIR AND EXISTS "${COREMRE_PACKAGE_DIR}/cmake/UseCoremre.cmake")
  include("${COREMRE_PACKAGE_DIR}/cmake/UseCoremre.cmake")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/engine/coremre/CMakeLists.txt")
  add_subdirectory(engine/coremre)
else()
  message(FATAL_ERROR "Missing coremre package")
endif()
add_subdirectory(.vxpe/platform/resources resources)
add_subdirectory(.vxpe/platform/mreapi mreapi)
add_subdirectory(.vxpe/platform/run run)
add_subdirectory(.vxpe/platform/main main)
